Service de Génération de Token/PIN // src/deposit/deposit.service.ts import { Injectable, Inject } from '@nestjs/common'; import { createHmac } from 'crypto'; import * as argon2 from 'argon2';
@Injectable() export class DepositService { constructor( @Inject('ARGON2') private readonly argon2: argon2.Argon2id, @Inject('PEPPER') private readonly pepper: string, ) {}
async generateTokenAndPin(): Promise<{ token: string; pin: string; tokenHash: string; pinHash: string }> { const token = this.generateRandomToken(); const pin = this.generateRandomPin();
const tokenHash = this.hashToken(token);  
const pinHash = await this.hashPin(pin);

return { token, pin, tokenHash, pinHash };  
}
private generateRandomToken(): string { return createHmac('sha256', this.pepper) .update(crypto.randomBytes(32)) .digest('base64url'); }
private generateRandomPin(): string { return crypto.randomInt(10000000, 99999999).toString().padStart(8, '0'); }
private hashToken(token: string): string { return createHmac('sha256', this.pepper).update(token).digest('hex'); }
private async hashPin(pin: string): Promise<string> { return this.argon2.hash(pin); }
async verifyPin(pinHash: string, pin: string): Promise<boolean> { return this.argon2.verify(pinHash, pin); } }
